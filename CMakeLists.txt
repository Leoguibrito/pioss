cmake_minimum_required(VERSION 3.12)
project(PIOSS VERSION 1.1 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_definitions(-D_POSIX_C_SOURCE=200809L)
option(PIOSS_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
if(PIOSS_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -fsanitize=address,undefined -no-pie")
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
include(CTest)
execute_process(COMMAND git rev-parse --short HEAD WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
                OUTPUT_VARIABLE PIOSS_REV OUTPUT_STRIP_TRAILING_WHITESPACE)
file(GLOB_RECURSE PIOSS_SOURCES src/*.c src/*.h lib/*.c lib/*.h)
list(SORT PIOSS_SOURCES)
set(PIOSS_CONTENT "")
foreach(source IN LISTS PIOSS_SOURCES)
  file(SHA256 ${source} digest)
  string(APPEND PIOSS_CONTENT "${digest}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${source})
endforeach()
string(SHA256 PIOSS_SOURCE_HASH "${PIOSS_CONTENT}")
configure_file(src/include/version.h.in generated/version.h @ONLY)
include_directories(${CMAKE_BINARY_DIR}/generated)
add_subdirectory(lib)
add_subdirectory(src)
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt")
  add_subdirectory(tests)
endif()
